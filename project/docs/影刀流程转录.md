## 5. read_csv 代码流程

> 影刀应用中的 `read_csv` 仍是早期版本（`iterrows` + 子串匹配），
> 仓库的 `project/expense_breakdown.py` 是后续重构版（编码查表 + 缓存 + 补规则）。
> 仓库版已用合成数据核验通过；以下为画布内实际源码，已按公开仓库词表替换路径与科目名。

```python
"""
支出拆解自动化 - Python 版本
对应影刀流程：
   收集匹配名称 → 构建字典
  ② 读取 CSV 源文件
  ③ 逐行匹配 + 金额累加
  ④ 返回字典 → 影刀写入 Excel
"""

import pandas as pd
import re
import xbot

BASE_DIR = r'<项目目录>'
CSV_PATH = BASE_DIR + r'\作业素材\源文件\demo_account_202512_账务明细.csv'


# ── 步骤1：构建匹配字典 ──
def build_rule_dict(args):
    rule_map = args
    sorted_keys = sorted(rule_map.keys(), key=len, reverse=True)
    xbot.logging.info(f'匹配规则数: {len(sorted_keys)}')
    return sorted_keys


# ── 步骤2：读取CSV ──
def read_csv_source(csv_path):
    df = pd.read_csv(csv_path, encoding='gbk', skiprows=4, dtype=str)
    df.columns = [c.strip() for c in df.columns]
    xbot.logging.info(f'列名: {df.columns.tolist()}')
    xbot.logging.info(f'CSV总行数: {len(df)}')
    return df


# ── 步骤3：逐行匹配 + 累加 ──
# 列号（0-indexed）：E=收入, F=支出, J=备注, O=业务描述
COL_INCOME  = 4   # 收入金额（+元）
COL_EXPENSE = 5   # 支出金额（-元）
COL_REMARK  = 9   # 备注
COL_DESC    = 14  # 业务描述

def safe_float(val):
    if val is None:
        return 0.0
    s = str(val).strip()
    if s in ('', '0.00', 'nan', 'None'):
        return 0.0
    try:
        return float(s)
    except:
        return 0.0

def clean_desc(desc):
    """去掉英文/中文括号及括号内内容（仅用于第二层normalize兜底）"""
    return re.sub(r'[（(][^）)]*[）)]', '', desc).strip()

def normalize(text):
    """去掉括号内容 → 去掉编码前缀 → 去掉数字和分隔符，用于兜底匹配"""
    cleaned = text
    cleaned = re.sub(r'[（(][^）)]*[）)]', '', cleaned)  # 去掉括号内容
    cleaned = re.sub(r'^\d+[A-Za-z]*\|', '', cleaned)
    cleaned = re.sub(r'\d+', '', cleaned)
    return cleaned.replace('-', '').replace('_', '').replace('：', '').replace(',', '').replace('，', '').replace('.', '').replace(' ', '')

def extract_text_after_pipe(text):
    """提取 | 后面的文字部分（不含编码前缀）"""
    if '|' in text:
        return text.split('|', 1)[1]
    return text

def match_desc(desc, sorted_keys, summary, amount):
    """尝试匹配，返回是否成功"""
    for kw in sorted_keys:
        if kw in desc:
            summary[kw] = summary.get(kw, 0) + amount
            return True
    return False

def match_and_accumulate(df, sorted_keys):
    summary = {}
    unmatched_descs = []
    total = 0.0
    fallback_count = 0

    # 预计算：normalize后的关键词（用于第二层兜底）
    normalized_keys = [(normalize(kw), kw) for kw in sorted_keys]

    # 预计算：提取|后面的文字部分（用于第三层兜底）
    text_keys = [(extract_text_after_pipe(kw).replace('-', '').replace('_', ''), kw) for kw in sorted_keys]

    for idx, (_, row) in enumerate(df.iterrows()):
        desc = (str(row.iloc[COL_DESC] or '') + str(row.iloc[COL_REMARK] or '')).strip()
        income  = safe_float(row.iloc[COL_INCOME])
        expense = safe_float(row.iloc[COL_EXPENSE])
        amount  = income + expense
        total += amount

        # 第一层：原样精确匹配
        if match_desc(desc, sorted_keys, summary, amount):
            continue

        if amount == 0:
            continue

        # 第二层：去掉括号和连字符后匹配
        normalized_desc = normalize(desc)
        matched = False
        for norm_kw, original_kw in normalized_keys:
            if norm_kw in normalized_desc:
                summary[original_kw] = summary.get(original_kw, 0) + amount
                fallback_count += 1
                matched = True
                break
        if matched:
            continue

        # 第三层：提取|后面的文字，忽略编码前缀差异
        desc_text = extract_text_after_pipe(desc).replace('-', '').replace('_', '')
        for text_kw, original_kw in text_keys:
            if text_kw in desc_text:
                summary[original_kw] = summary.get(original_kw, 0) + amount
                fallback_count += 1
                matched = True
                break
        if matched:
            continue

        # 第四层：特殊关键词匹配（平台补贴、推广联盟等）
        if '平台补贴' in desc:
            if amount < 0:
                summary['1200001|其他支出-平台补贴预收'] = summary.get('1200001|其他支出-平台补贴预收', 0) + amount
            else:
                summary['1200002T|其他收入-平台补贴激励前返'] = summary.get('1200002T|其他收入-平台补贴激励前返', 0) + amount
            fallback_count += 1
            continue

        if '推广联盟' in desc:
            summary['推广联盟佣金代扣'] = summary.get('推广联盟佣金代扣', 0) + amount
            fallback_count += 1
            continue

        if '营销托管' in desc and '代扣' in desc:
            summary['营销托管-平台营销托管服务费-扣款'] = summary.get('营销托管-平台营销托管服务费-扣款', 0) + amount
            fallback_count += 1
            continue

        if '营销托管' in desc and '退款' in desc:
            summary['营销托管新客计划'] = summary.get('营销托管新客计划', 0) + amount
            fallback_count += 1
            continue

        if '平台转账' in desc:
            summary['提现'] = summary.get('提现', 0) + amount
            fallback_count += 1
            continue

        if '本地退赔款' in desc:
            summary['提现'] = summary.get('提现', 0) + amount
            fallback_count += 1
            continue

        if desc.strip() == '转账':
            summary['提现'] = summary.get('提现', 0) + amount
            fallback_count += 1
            continue

        unmatched_descs.append((idx, desc, amount))

    if unmatched_descs:
        xbot.logging.info(f'未匹配({len(unmatched_descs)}条):')
        for lineno, d, amt in unmatched_descs[:30]:
            xbot.logging.info(f'  行{lineno}: 金额={amt:.2f} 描述={d}')
    else:
        xbot.logging.info('全部匹配成功！')

    return summary, total


# ── 主入口 ─
def main(args):
    sorted_keys = build_rule_dict(args)
    df = read_csv_source(CSV_PATH)
    summary, total = match_and_accumulate(df, sorted_keys)

    # 确保所有分类都有key（即使金额为0），避免影刀取不到key报错
    full_summary = {}
    for kw in sorted_keys:
        full_summary[kw] = summary.get(kw, 0.0)

    return {'summary': full_summary, 'total': total}
```
